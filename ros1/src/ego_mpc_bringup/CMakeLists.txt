cmake_minimum_required(VERSION 3.16)
project(ego_mpc_bringup)
find_package(catkin REQUIRED COMPONENTS roscpp nav_msgs geometry_msgs std_msgs quadrotor_msgs traj_utils)
catkin_package()
add_executable(navigation_node src/navigation_node.cpp)
target_compile_features(navigation_node PRIVATE cxx_std_14)
target_include_directories(navigation_node PRIVATE ../../../safety/include ${catkin_INCLUDE_DIRS})
target_link_libraries(navigation_node ${catkin_LIBRARIES})
add_dependencies(navigation_node ${catkin_EXPORTED_TARGETS})
install(TARGETS navigation_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
 add_executable(check_command_gate ../../../safety/tests/check_command_gate.cpp)
 target_include_directories(check_command_gate PRIVATE ../../../safety/include)
 add_test(NAME command_gate COMMAND check_command_gate)
endif()

if(CATKIN_ENABLE_TESTING)
 add_executable(runtime_pipeline ../../../tests/runtime_pipeline.cpp)
 target_include_directories(runtime_pipeline PRIVATE ${catkin_INCLUDE_DIRS})
 target_link_libraries(runtime_pipeline ${catkin_LIBRARIES})
 add_dependencies(runtime_pipeline ${catkin_EXPORTED_TARGETS})
endif()
