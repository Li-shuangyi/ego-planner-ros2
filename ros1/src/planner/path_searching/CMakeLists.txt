cmake_minimum_required(VERSION 3.16)
project(path_searching)
find_package(catkin REQUIRED COMPONENTS roscpp plan_env)
find_package(Eigen3 REQUIRED)
catkin_package(INCLUDE_DIRS include LIBRARIES path_searching CATKIN_DEPENDS roscpp plan_env DEPENDS Eigen3)
add_library(path_searching src/dyn_a_star.cpp)
target_compile_features(path_searching PUBLIC cxx_std_14)
target_include_directories(path_searching PUBLIC include ${catkin_INCLUDE_DIRS})
target_link_libraries(path_searching PUBLIC ${catkin_LIBRARIES} Eigen3::Eigen)
add_dependencies(path_searching ${catkin_EXPORTED_TARGETS})
install(TARGETS path_searching ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
