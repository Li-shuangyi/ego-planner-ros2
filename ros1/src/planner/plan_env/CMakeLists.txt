cmake_minimum_required(VERSION 3.16)
project(plan_env)
find_package(catkin REQUIRED COMPONENTS roscpp nav_msgs std_msgs tf2_ros)
find_package(Eigen3 REQUIRED)
catkin_package(INCLUDE_DIRS include LIBRARIES plan_env CATKIN_DEPENDS roscpp nav_msgs std_msgs tf2_ros DEPENDS Eigen3)
add_library(plan_env src/grid_map.cpp)
target_compile_features(plan_env PUBLIC cxx_std_14)
target_include_directories(plan_env PUBLIC include ${catkin_INCLUDE_DIRS})
target_link_libraries(plan_env PUBLIC ${catkin_LIBRARIES} Eigen3::Eigen)
add_dependencies(plan_env ${catkin_EXPORTED_TARGETS})
install(TARGETS plan_env ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  add_executable(check_shared_grid tests/check_shared_grid.cpp)
  target_compile_options(check_shared_grid PRIVATE -fno-access-control -UNDEBUG)
  target_link_libraries(check_shared_grid plan_env)
endif()
