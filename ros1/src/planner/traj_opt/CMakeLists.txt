cmake_minimum_required(VERSION 3.16)
project(traj_opt)
find_package(catkin REQUIRED COMPONENTS roscpp plan_env path_searching traj_utils)
find_package(Eigen3 REQUIRED)
catkin_package(INCLUDE_DIRS include LIBRARIES traj_opt CATKIN_DEPENDS roscpp plan_env path_searching traj_utils DEPENDS Eigen3)
add_library(traj_opt src/poly_traj_optimizer.cpp)
target_compile_features(traj_opt PUBLIC cxx_std_14)
target_include_directories(traj_opt PUBLIC include ${catkin_INCLUDE_DIRS})
target_link_libraries(traj_opt PUBLIC ${catkin_LIBRARIES} Eigen3::Eigen)
add_dependencies(traj_opt ${catkin_EXPORTED_TARGETS})
install(TARGETS traj_opt ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
