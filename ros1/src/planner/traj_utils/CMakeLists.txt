cmake_minimum_required(VERSION 3.16)
project(traj_utils)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs geometry_msgs visualization_msgs message_generation)
find_package(Eigen3 REQUIRED)
add_message_files(FILES DataDisp.msg PolyTraj.msg PlannerCommand.msg MINCOTraj.msg)
generate_messages(DEPENDENCIES std_msgs geometry_msgs)
catkin_package(INCLUDE_DIRS include LIBRARIES traj_utils CATKIN_DEPENDS roscpp std_msgs geometry_msgs visualization_msgs message_runtime DEPENDS Eigen3)
add_library(traj_utils src/planning_visualization.cpp)
target_compile_features(traj_utils PUBLIC cxx_std_14)
target_include_directories(traj_utils PUBLIC include ${catkin_INCLUDE_DIRS})
target_link_libraries(traj_utils PUBLIC ${catkin_LIBRARIES} Eigen3::Eigen)
add_dependencies(traj_utils ${catkin_EXPORTED_TARGETS} ${${PROJECT_NAME}_EXPORTED_TARGETS})
install(TARGETS traj_utils ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
