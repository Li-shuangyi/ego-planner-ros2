cmake_minimum_required(VERSION 3.22)
project(ego_mpc_bringup)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp nav_msgs geometry_msgs sensor_msgs std_msgs quadrotor_msgs traj_utils)
 find_package(${dep} REQUIRED)
endforeach()
add_executable(navigation_node src/navigation_node.cpp)
target_compile_features(navigation_node PRIVATE cxx_std_17)
target_include_directories(navigation_node PRIVATE ../../../safety/include)
ament_target_dependencies(navigation_node rclcpp nav_msgs geometry_msgs sensor_msgs std_msgs quadrotor_msgs traj_utils)
install(TARGETS navigation_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 enable_testing()
 add_executable(check_command_gate ../../../safety/tests/check_command_gate.cpp)
 target_include_directories(check_command_gate PRIVATE ../../../safety/include)
 add_test(NAME command_gate COMMAND check_command_gate)
endif()

if(BUILD_TESTING)
 add_executable(runtime_pipeline ../../../tests/runtime_pipeline.cpp)
 target_compile_definitions(runtime_pipeline PRIVATE EGO_MPC_ROS2)
 ament_target_dependencies(runtime_pipeline rclcpp nav_msgs geometry_msgs traj_utils)
 install(TARGETS runtime_pipeline RUNTIME DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
